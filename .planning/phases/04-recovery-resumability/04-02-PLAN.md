       export async function failTask(opts: FailTaskOpts): Promise<FailTaskResult> {
         const { taskId, workflowId, projectDir, agentId, failureCategory, failureReason, sessionKey } = opts;

         // Fire observability hook
         await triggerInternalHook(
           createInternalHookEvent("project", "task-failed", sessionKey ?? "", {
             taskId, workflowId, projectDir, agentId, failureCategory, failureReason,
           }),
         );

         // Execute recovery chain
         const result = await executeRecoveryStrategy({
           taskId, workflowId, projectDir, agentId, failureCategory, failureReason, sessionKey,
         });

         log.info("Task failure processed", { taskId, outcome: result.outcome, failureCategory });

         return result;
       }
